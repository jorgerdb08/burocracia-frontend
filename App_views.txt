// ── Subir Archivos ─────────────────────────────────────────────────────────
function ViewSubir({ onSubido }: { onSubido: (id: string) => void }) {
  const [files, setFiles] = useState<File[]>([])
  const [uploading, setUploading] = useState(false)
  const [err, setErr] = useState('')

  const handleUpload = async () => {
    if (files.length === 0) return
    setUploading(true)
    setErr('')
    try {
      const fd = new FormData()
      files.forEach(f => fd.append('files', f))
      const res = await fetch('/api/caso/upload', { method: 'POST', body: fd })
      const data = await res.json()
      if (data.error) throw new Error(data.error)
      onSubido(data.caso_id)
    } catch (e: any) {
      setErr(e.message)
    } finally {
      setUploading(false)
    }
  }

  return (
    <div className="bg-white dark:bg-[#0a0a0a] p-6 border border-zinc-200 dark:border-zinc-700 shadow-sm">
      <SecHeading title="Aportar Justificantes" subtitle="Sube los PDFs o TXTs de la comisión de servicio para procesarlos automáticamente." />
      
      <div className="border-2 border-dashed border-zinc-300 dark:border-zinc-600 p-8 text-center rounded-sm bg-zinc-50 dark:bg-zinc-900 mb-6">
        <input type="file" multiple className="mx-auto block" onChange={e => setFiles(Array.from(e.target.files || []))} />
        <p className="text-zinc-500 dark:text-zinc-400 text-sm mt-3">Archivos seleccionados: {files.length}</p>
      </div>

      {err && <AlertErr msg={err} />}

      <div className="flex justify-end">
        <Btn onClick={handleUpload} disabled={files.length === 0 || uploading}>
          {uploading ? <Loader2 size={16} className="animate-spin" /> : <Play size={16} />}
          Procesar nuevos justificantes
        </Btn>
      </div>
    </div>
  )
}

// ── Procesar Caso ──────────────────────────────────────────────────────────
function ViewProcesar({ initialCaso, templateId }: { initialCaso?: string, templateId?: string }) {
  const [casos, setCasos]     = useState<{id: string, tipo: string}[]>([])
  const [casoSel, setCasoSel] = useState(initialCaso ?? '')
  const sinLlm = Boolean(casoSel && casoSel.startsWith('caso_') && !casoSel.startsWith('caso_usuario_'))
  const [running, setRunning] = useState(false)
  const [result, setResult]   = useState<CasoResult | null>(null)
  const [error, setError]     = useState('')
  const [tipoProyecto, setTipoProyecto] = useState('')
  const [modalidad, setModalidad] = useState('')
  const resultRef             = useRef<HTMLDivElement>(null)

  useEffect(() => {
    fetch('/api/casos').then(r => r.json()).then(d => setCasos(d.casos)).catch(() => {})
  }, [])

  const casosFiltrados = useMemo(() => {
    if (!templateId) return casos
    return casos.filter(c => c.tipo === templateId)
  }, [casos, templateId])

  useEffect(() => {
    if (result && resultRef.current)
      resultRef.current.scrollIntoView({ behavior: 'smooth', block: 'start' })
  }, [result])

  const [files, setFiles] = useState<File[]>([])

  const ejecutar = useCallback(async () => {
    let targetCaso = casoSel;
    setRunning(true); setError(''); setResult(null)
    try {
      if (!targetCaso) {
        if (files.length === 0) throw new Error("Debe subir archivos para justificar el gasto.")
        const fd = new FormData()
        files.forEach(f => fd.append('files', f))
        const res = await fetch('/api/caso/upload', { method: 'POST', body: fd })
        const data = await res.json()
        if (data.error) throw new Error(data.error)
        targetCaso = data.caso_id
        setCasoSel(targetCaso)
      }

      const s = await fetch('/api/caso/iniciar', {
        method: 'POST', headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ caso_id: targetCaso, sin_llm: sinLlm, tipo_proyecto: tipoProyecto }),
      })
      const { task_id } = await s.json()
      setResult(await pollTask<CasoResult>(task_id))
    } catch (e: any) { setError(e.message) }
    finally { setRunning(false) }
  }, [casoSel, files, sinLlm, tipoProyecto])

  return (
    <div className="space-y-6">
      <div className="bg-white dark:bg-[#0a0a0a] p-6 border border-zinc-200 dark:border-zinc-700 shadow-sm">
        <SecHeading title="Tramitación de Justificación" subtitle="Módulo de extracción y cálculo automatizado para liquidaciones" />

        <div className="bg-zinc-50 dark:bg-zinc-900 p-4 border border-zinc-200 dark:border-zinc-700 rounded-sm">
          <div className="flex flex-col md:flex-row gap-4 items-end">
            {!initialCaso ? (
              <div className="flex-1 flex flex-col">
                <label className="block text-sm font-bold text-zinc-700 dark:text-zinc-200 mb-2">Expediente a procesar:</label>
                <div className="space-y-2">
                  <select value={casoSel}
                    onChange={e => { setCasoSel(e.target.value); setResult(null); setError('') }}
                    className="w-full px-3 py-1.5 text-sm bg-white dark:bg-[#0a0a0a] border border-zinc-300 dark:border-zinc-600 rounded-sm focus:outline-none focus:border-zinc-500 text-zinc-700 dark:text-zinc-200">
                    <option value="">➕ Subir nuevos archivos desde mi equipo...</option>
                    {casosFiltrados.length > 0 && (
                      <optgroup label="Expedientes cargados en el servidor">
                        {casosFiltrados.map(c => <option key={c.id} value={c.id}>{c.id} ({c.tipo})</option>)}
                      </optgroup>
                    )}
                  </select>
                  {!casoSel && (
                    <input type="file" multiple onChange={e => setFiles(Array.from(e.target.files || []))}
                      className="w-full px-3 py-1.5 text-sm bg-zinc-50 dark:bg-[#0a0a0a] border border-zinc-200 dark:border-zinc-700 rounded-sm focus:outline-none focus:border-zinc-500 text-zinc-700 dark:text-zinc-200 file:mr-4 file:py-1 file:px-3 file:rounded-sm file:border-0 file:text-xs file:font-semibold file:bg-zinc-200 dark:file:bg-zinc-900 file:text-zinc-700 dark:file:text-zinc-300 hover:file:bg-slate-300 dark:hover:file:bg-zinc-900 cursor-pointer" />
                  )}
                </div>
              </div>
            ) : (
              <div className="flex-1">
                <label className="block text-sm font-bold text-zinc-700 dark:text-zinc-200 mb-2">Expediente Seleccionado:</label>

}


// ── Grafo ──────────────────────────────────────────────────────────────────
function ViewGrafo() {
  const [running, setRunning] = useState(false)
  const [mermaid, setMermaid] = useState('')
  const [error, setError]     = useState('')

  const generar = useCallback(async () => {
    setRunning(true); setError(''); setMermaid('')
    try {
      const d = await (await fetch('/api/grafo')).json()
      if (!d.ok) throw new Error(d.error)
      setMermaid(d.mermaid)
    } catch (e: any) { setError(e.message) }
    finally { setRunning(false) }
  }, [])

  const src = mermaid
    ? `<!DOCTYPE html><html><head><style>html,body{margin:0;padding:20px;background:#fff;font-family:Inter,sans-serif}.mermaid svg{max-width:100%;height:auto}</style></head><body><pre class="mermaid">${mermaid}</pre><script type="module">import mermaid from 'https://cdn.jsdelivr.net/npm/mermaid@10/dist/mermaid.esm.min.mjs';mermaid.initialize({startOnLoad:true,theme:'neutral'});<\/script></body></html>`
    : ''

  return (
    <div className="space-y-4">
      <div className="flex justify-between items-center bg-zinc-50 dark:bg-zinc-900 border border-zinc-200 dark:border-zinc-700 p-4 rounded-sm">
        <div>
          <h3 className="text-base font-bold text-zinc-800 dark:text-zinc-100">Auditoría de Arquitectura (LangGraph)</h3>
          <p className="text-sm text-zinc-500 dark:text-zinc-400">Inspeccione el flujo de decisiones tomado por el agente para este trámite.</p>
        </div>
        <Btn size="sm" onClick={generar} disabled={running}>
          {running ? <><Loader2 size={14} className="animate-spin" /> Dibujando...</> : <><GitBranch size={14} /> Generar Grafo</>}
        </Btn>
      </div>

      {!running && !mermaid && !error && (
        <div className="flex flex-col items-center justify-center py-12 bg-zinc-50 dark:bg-zinc-900 border border-zinc-200 dark:border-zinc-700 text-zinc-500 dark:text-zinc-400 rounded-sm">
          <GitBranch size={40} className="mb-4 text-zinc-300" />
          <p className="text-sm font-medium">Pulse el botón superior para auditar la estructura de pasos ejecutada.</p>
        </div>
      )}
      {running && (
        <div className="flex items-center justify-center gap-3 py-16 text-zinc-500 dark:text-zinc-400 bg-zinc-50 dark:bg-zinc-900 border border-zinc-200 dark:border-zinc-700">
          <Loader2 size={24} className="animate-spin text-zinc-400" />
          <span className="text-base">Generando representación visual del autómata...</span>
        </div>
      )}
      {error && <AlertErr msg={error} />}
      {mermaid && (
        <div className="space-y-4 mt-6">
          <div className="border border-zinc-200 dark:border-zinc-700 bg-white dark:bg-[#0a0a0a]">
            <iframe srcDoc={src} className="w-full h-[600px] border-none" title="LangGraph arquitectura" />
          </div>
        </div>
      )}
    </div>
  )
}
